import { PrismaClient } from "@prisma/client";

type Vec3 = [number, number, number];

function sub(a: Vec3, b: Vec3): Vec3 {
  return [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
}

function cross(a: Vec3, b: Vec3): Vec3 {
  return [
    a[1] * b[2] - a[2] * b[1],
    a[2] * b[0] - a[0] * b[2],
    a[0] * b[1] - a[1] * b[0],
  ];
}

function dot(a: Vec3, b: Vec3): number {
  return a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
}

function normalize(v: Vec3): Vec3 {
  const len = Math.hypot(v[0], v[1], v[2]) || 1;
  return [v[0] / len, v[1] / len, v[2] / len];
}

function fmt(n: number): string {
  return Number(n.toFixed(6)).toString();
}

/**
 * Construit un fichier STL ASCII valide à partir d'une liste de triangles
 * et d'un point intérieur (pour garantir l'orientation sortante des normales).
 */
function buildAsciiStl(
  solidName: string,
  triangles: [Vec3, Vec3, Vec3][],
  interiorPoint: Vec3
): string {
  const lines: string[] = [`solid ${solidName}`];

  for (const [p1, p2, p3] of triangles) {
    let v1 = p1;
    let v2 = p2;
    let v3 = p3;

    let n = cross(sub(v2, v1), sub(v3, v1));
    const outwardRef = sub(v1, interiorPoint);
    if (dot(n, outwardRef) < 0) {
      [v2, v3] = [v3, v2];
      n = cross(sub(v2, v1), sub(v3, v1));
    }
    const unit = normalize(n);

    lines.push(`  facet normal ${fmt(unit[0])} ${fmt(unit[1])} ${fmt(unit[2])}`);
    lines.push("    outer loop");
    lines.push(`      vertex ${fmt(v1[0])} ${fmt(v1[1])} ${fmt(v1[2])}`);
    lines.push(`      vertex ${fmt(v2[0])} ${fmt(v2[1])} ${fmt(v2[2])}`);
    lines.push(`      vertex ${fmt(v3[0])} ${fmt(v3[1])} ${fmt(v3[2])}`);
    lines.push("    endloop");
    lines.push("  endfacet");
  }

  lines.push(`endsolid ${solidName}`);
  return lines.join("\n") + "\n";
}

function createCubeStl(size = 24): string {
  const h = size / 2;
  const b0: Vec3 = [-h, -h, 0];
  const b1: Vec3 = [h, -h, 0];
  const b2: Vec3 = [h, h, 0];
  const b3: Vec3 = [-h, h, 0];
  const t0: Vec3 = [-h, -h, size];
  const t1: Vec3 = [h, -h, size];
  const t2: Vec3 = [h, h, size];
  const t3: Vec3 = [-h, h, size];

  const quads: [Vec3, Vec3, Vec3, Vec3][] = [
    [b0, b1, b2, b3], // bottom
    [t0, t1, t2, t3], // top
    [b0, b1, t1, t0], // front
    [b1, b2, t2, t1], // right
    [b2, b3, t3, t2], // back
    [b3, b0, t0, t3], // left
  ];

  const triangles: [Vec3, Vec3, Vec3][] = [];
  for (const [a, b, c, d] of quads) {
    triangles.push([a, b, c], [a, c, d]);
  }

  return buildAsciiStl("cube_calibration_24mm", triangles, [0, 0, size / 2]);
}

function createPyramidStl(base = 32, height = 28): string {
  const h = base / 2;
  const b0: Vec3 = [-h, -h, 0];
  const b1: Vec3 = [h, -h, 0];
  const b2: Vec3 = [h, h, 0];
  const b3: Vec3 = [-h, h, 0];
  const apex: Vec3 = [0, 0, height];

  const triangles: [Vec3, Vec3, Vec3][] = [
    [b0, b1, b2],
    [b0, b2, b3],
    [b0, b1, apex],
    [b1, b2, apex],
    [b2, b3, apex],
    [b3, b0, apex],
  ];

  return buildAsciiStl("pyramide_support", triangles, [0, 0, height / 4]);
}

function createHexPrismStl(radius = 16, height = 22): string {
  const bottom: Vec3[] = [];
  const top: Vec3[] = [];
  for (let i = 0; i < 6; i++) {
    const angle = (Math.PI / 3) * i;
    const x = radius * Math.cos(angle);
    const y = radius * Math.sin(angle);
    bottom.push([x, y, 0]);
    top.push([x, y, height]);
  }

  const bottomCenter: Vec3 = [0, 0, 0];
  const topCenter: Vec3 = [0, 0, height];
  const triangles: [Vec3, Vec3, Vec3][] = [];

  for (let i = 0; i < 6; i++) {
    const next = (i + 1) % 6;
    // Bottom cap
    triangles.push([bottomCenter, bottom[i], bottom[next]]);
    // Top cap
    triangles.push([topCenter, top[i], top[next]]);
    // Side quad
    triangles.push([bottom[i], bottom[next], top[next]]);
    triangles.push([bottom[i], top[next], top[i]]);
  }

  return buildAsciiStl("entretoise_hexagonale", triangles, [0, 0, height / 2]);
}

import { NextRequest, NextResponse } from "next/server";
import path from "node:path";
import { getOrderById } from "@/services/orders";
import { analyzeStlWithPrusaSlicer } from "@/lib/slicer";

export async function POST(
  request: NextRequest,
  context: { params: Promise<{ id: string }> }
) {
  try {
    const { id } = await context.params;
    const order = await getOrderById(id);

    if (!order) {
      return NextResponse.json(
        { error: "Demande introuvable." },
        { status: 404 }
      );
    }

    const stlBaseDir = path.resolve(process.cwd(), "data", "stl");
    const resolvedPath = path.resolve(process.cwd(), order.stlPath);

    if (
      !resolvedPath.startsWith(stlBaseDir + path.sep) &&
      resolvedPath !== stlBaseDir
    ) {
      return NextResponse.json(
        { error: "Chemin de fichier STL non autorisé." },
        { status: 403 }
      );
    }

    const body = await request.json().catch(() => ({}));
    const analysis = await analyzeStlWithPrusaSlicer({
      stlAbsolutePath: resolvedPath,
      infillPercent:
        typeof body.infillPercent === "number" ? body.infillPercent : 15,
      supports: Boolean(body.supports),
    });

    return NextResponse.json({ analysis });
  } catch (error) {
    return NextResponse.json(
      {
        error:
          error instanceof Error
            ? error.message
            : "Erreur lors du tranchage PLA avec PrusaSlicer.",
      },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import path from "node:path";
import { getOrderById, updateOrderStatus } from "@/services/orders";
import { getActiveSlicerSessionInfo } from "@/lib/slicer-embed";
import {
  findLatestExportedGcodeSince,
  parsePrusaGcodeMetadata,
} from "@/lib/gcode-parser";

export const dynamic = "force-dynamic";

/**
 * Détecte en temps réel tout fichier `.gcode` ou `.bgcode` exporté par l'opérateur
 * depuis ses propres changements dans le PrusaSlicer intégré au centre,
 * et renvoie automatiquement le poids PLA exact + paramètres.
 */
export async function GET(
  _request: NextRequest,
  context: { params: Promise<{ id: string }> }
) {
  try {
    const { id } = await context.params;
    const order = await getOrderById(id);
    if (!order) {
      return NextResponse.json(
        { error: "Demande introuvable." },
        { status: 404 }
      );
    }

    const session = getActiveSlicerSessionInfo();

    // Si cette commande est actuellement ouverte dans PrusaSlicer, on cherche
    // un export effectué après l'ouverture de la pièce.
    if (session.activeOrderId === id) {
      const latest = await findLatestExportedGcodeSince(
        session.sessionOpenedAtMs
      );
      if (latest) {
        if (order.gcodeFilename !== latest.filename) {
          await updateOrderStatus(id, order.status, {
            gcodeFilename: latest.filename,
          }).catch(() => {});
        }
        return NextResponse.json({
          synced: true,
          slice: latest,
        });
      }
    }

    // Sinon, si la commande possède déjà un gcodeFilename enregistré, on lit ses métadonnées
    if (order.gcodeFilename) {
      const existingPath = path.resolve(
        process.cwd(),
        "data",
        "gcode",
        order.gcodeFilename
      );
      const parsed = await parsePrusaGcodeMetadata(existingPath);
      if (parsed) {
        return NextResponse.json({
          synced: true,
          slice: parsed,
        });
      }
    }

    return NextResponse.json({ synced: false, slice: null });
  } catch (error) {
    return NextResponse.json(
      {
        synced: false,
        error: error instanceof Error ? error.message : "Erreur sync G-code",
      },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import fs from "node:fs/promises";
import path from "node:path";
import { getOrderById } from "@/services/orders";

export async function GET(
  request: NextRequest,
  context: { params: Promise<{ id: string }> }
) {
  try {
    const { id } = await context.params;
    const order = await getOrderById(id);

    if (!order) {
      return NextResponse.json(
        { error: "Demande introuvable." },
        { status: 404 }
      );
    }

    const stlBaseDir = path.resolve(process.cwd(), "data", "stl");
    const resolvedPath = path.resolve(process.cwd(), order.stlPath);

    if (
      !resolvedPath.startsWith(stlBaseDir + path.sep) &&
      resolvedPath !== stlBaseDir
    ) {
      return NextResponse.json(
        { error: "Chemin de fichier STL non autorisé." },
        { status: 403 }
      );
    }

    const fileBuffer = await fs.readFile(resolvedPath);
    const isDownload = request.nextUrl.searchParams.get("download") === "1";

    const safeDownloadName = order.stlFilename.replace(/["\r\n]/g, "_");

    return new NextResponse(new Uint8Array(fileBuffer), {
      status: 200,
      headers: {
        "Content-Type": "model/stl",
        "Content-Length": String(fileBuffer.byteLength),
        "Content-Disposition": isDownload
          ? `attachment; filename="${safeDownloadName}"`
          : `inline; filename="${safeDownloadName}"`,
        "Cache-Control": "no-store",
      },
    });
  } catch (error) {
    return NextResponse.json(
      {
        error:
          error instanceof Error
            ? `Impossible de lire le fichier STL : ${error.message}`
            : "Fichier STL introuvable sur le disque.",
      },
      { status: 404 }
    );
  }
}

export async function PATCH(
  request: NextRequest,
  context: { params: Promise<{ id: string }> }
) {
  try {
    const { id } = await context.params;
    const body = await request.json();

    if (body.action === "decision") {
      const verdict = body.verdict as OrderVerdict;
      if (!ORDER_VERDICTS.includes(verdict)) {
        return NextResponse.json(
          { error: "Verdict invalide." },
          { status: 400 }
        );
      }

      const order = await updateOrderDecision(id, {
        verdict,
        weightGrams:
          body.weightGrams !== undefined && body.weightGrams !== null
            ? Number(body.weightGrams)
            : null,
        operatorComment: body.operatorComment ?? null,
        needsDesignerMeeting: Boolean(body.needsDesignerMeeting),
        improvementAction: body.improvementAction,
      });

      return NextResponse.json({ order });
    }

    if (body.action === "status") {
      const status = body.status as OrderStatus;
      if (!ORDER_STATUSES.includes(status)) {
        return NextResponse.json(
          { error: "Statut invalide." },
          { status: 400 }
        );
      }

      const order = await updateOrderStatus(id, status, {
        gcodeFilename: body.gcodeFilename,
        invoiced: body.invoiced,
      });

      if (status === "FINISHED" && isPrusaMockEnabled()) {
        await setMockPrinterState("FINISHED");
      }

      return NextResponse.json({ order });
    }

    return NextResponse.json(
      { error: "Action non reconnue (attendu: 'decision' ou 'status')." },
      { status: 400 }
    );
  } catch (error) {
    return NextResponse.json(
      {
        error:
          error instanceof Error
            ? error.message
            : "Erreur lors de la mise à jour de la demande.",
      },
      { status: 400 }
    );
  }
}

/**
 * Récupère le fichier .gcode ou .bgcode le plus récemment exporté depuis le PrusaSlicer intégré
 * dans ./data/gcode/ et le lance directement sur la Prusa MK4.
 */
export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    const orderId = String(body.orderId ?? "").trim();

    if (!orderId) {
      return NextResponse.json(
        { error: "Identifiant de la demande (orderId) manquant." },
        { status: 400 }
      );
    }

    const order = await getOrderById(orderId);
    if (!order) {
      return NextResponse.json(
        { error: "Demande introuvable." },
        { status: 404 }
      );
    }

    const gcodeDir = path.resolve(process.cwd(), "data", "gcode");
    await fs.mkdir(gcodeDir, { recursive: true });
    const entries = await fs.readdir(gcodeDir);

    let newest: { name: string; mtimeMs: number; fullPath: string } | null =
      null;

    for (const name of entries) {
      const ext = path.extname(name).toLowerCase();
      if (ext !== ".gcode" && ext !== ".bgcode") continue;
      const fullPath = path.join(gcodeDir, name);
      const st = await fs.stat(fullPath).catch(() => null);
      if (!st || !st.isFile()) continue;
      if (!newest || st.mtimeMs > newest.mtimeMs) {
        newest = { name, mtimeMs: st.mtimeMs, fullPath };
      }
    }

    if (!newest) {
      return NextResponse.json(
        {
          error:
            "Aucun fichier .gcode ou .bgcode trouvé dans ./data/gcode/. Cliquez d'abord sur 'Exporter le G-code' dans PrusaSlicer au centre (ou glissez un fichier ci-dessous).",
        },
        { status: 404 }
      );
    }

    const fileBuffer = await fs.readFile(newest.fullPath);
    const result = await uploadGcodeAndStartPrint({
      orderId,
      originalFilename: newest.name,
      fileBuffer,
    });

    return NextResponse.json(result);
  } catch (error) {
    return NextResponse.json(
      {
        error:
          error instanceof Error
            ? error.message
            : "Erreur lors de l'envoi du G-code exporté.",
      },
      { status: 500 }
    );
  }

      });

      const data = await res.json();
      if (!res.ok) {
        throw new Error(data.error ?? "Erreur lors de l'enregistrement.");
      }
      onOrderUpdated(data.order);
      setFeedbackSuccess("Décision enregistrée avec succès.");
    } catch (err) {
      setFeedbackError(
        err instanceof Error ? err.message : "Erreur inattendue."
      );
    } finally {
      setSubmitting(false);
    }
  };

  const updateStatus = async (
    nextStatus: PrintOrder["status"],
    extra?: { invoiced?: boolean }
  ) => {
    setSubmitting(true);
    setFeedbackError(null);
    setFeedbackSuccess(null);
    try {
      const res = await fetch(`/api/orders/${order.id}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          action: "status",
          status: nextStatus,
          ...extra,
        }),
      });
      const data = await res.json();
      if (!res.ok) {
        throw new Error(data.error ?? "Impossible de mettre à jour le statut.");
      }
      onOrderUpdated(data.order);
      onPrinterAction();
    } catch (err) {
      setFeedbackError(
        err instanceof Error ? err.message : "Erreur inattendue."
      );
    } finally {
      setSubmitting(false);
    }
  };

  const handleSelectGcodeFile = (file: File | undefined) => {
    if (!file) return;
    const lower = file.name.toLowerCase();
    if (!lower.endsWith(".gcode") && !lower.endsWith(".bgcode")) {
      setFeedbackError(
        "Format invalide : veuillez sélectionner un fichier .gcode ou .bgcode."
      );
      return;
    }
    setFeedbackError(null);
    setGcodeFile(file);
  };

  const handleUploadAndPrint = async () => {
    if (!gcodeFile) {
      setFeedbackError(
        "Veuillez glisser un fichier .gcode ou .bgcode généré par PrusaSlicer."
      );
      return;
    }

    setSubmitting(true);
    setFeedbackError(null);
    setFeedbackSuccess(null);

    try {
      const formData = new FormData();
      formData.append("orderId", order.id);
      formData.append("file", gcodeFile);

      const res = await fetch("/api/printer/print", {
        method: "POST",
        body: formData,
      });
      const data = await res.json();
      if (!res.ok) {
        throw new Error(data.error ?? "Échec de l'envoi vers la Prusa.");
      }

      onOrderUpdated(data.order);
      onPrinterAction();
      setFeedbackSuccess(
        data.mock
          ? `Fichier sauvegardé (${data.filename}) et impression simulée démarrée.`
          : `Fichier envoyé sur votre Prusa MK4 (${data.filename}) et impression lancée !`
      );
    } catch (err) {
      setFeedbackError(
        err instanceof Error ? err.message : "Erreur lors de l'envoi."
      );
    } finally {
      setSubmitting(false);
    }
  };

  const handleSendLatestExportedGcode = async () => {
    setSubmitting(true);
    setFeedbackError(null);
    setFeedbackSuccess(null);
    try {
      const res = await fetch("/api/printer/print-latest", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ orderId: order.id }),
      });
      const data = await res.json();
      if (!res.ok) {
        throw new Error(
          data.error ?? "Aucun G-code récent trouvé dans ./data/gcode/."
        );
      }
      onOrderUpdated(data.order);
      onPrinterAction();
      setFeedbackSuccess(
        `G-code PrusaSlicer (${data.filename}) envoyé et lancé sur la Prusa !`
      );
    } catch (err) {
      setFeedbackError(
        err instanceof Error ? err.message : "Erreur lors de l'envoi."
      );
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <div className="flex flex-col h-full">
      <div className="px-4 py-3 border-b border-slate-200 bg-white">
        <div className="flex items-center justify-between">
          <h2 className="text-sm font-semibold text-slate-900">
            Décision & Actions
          </h2>
          <span className="text-[10px] font-semibold uppercase tracking-wider px-2 py-0.5 rounded bg-slate-100 text-slate-700 border border-slate-200">
            PLA Standard
          </span>
        </div>
        <p className="text-xs text-slate-500 mt-0.5">
          Étape actuelle :{" "}
          <span className="font-medium text-slate-700">{order.status}</span>
        </p>
      </div>

      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {feedbackError && (
          <div className="rounded-lg bg-rose-50 border border-rose-200 p-3 text-xs text-rose-700">
            {feedbackError}
          </div>
        )}

        {feedbackSuccess && (
          <div className="rounded-lg bg-emerald-50 border border-emerald-200 p-3 text-xs text-emerald-700">
            {feedbackSuccess}
          </div>
        )}

        {/* ==================== STATUT : TO_ANALYZE ==================== */}
        {order.status === "TO_ANALYZE" && (
          <form onSubmit={submitDecision} className="space-y-4">
            <div className="space-y-2">
              <label className="block text-xs font-semibold uppercase tracking-wider text-slate-500">
                Diagnostic d&apos;imprimabilité (PLA)
              </label>

              {/* Choix 1 : Imprimable (Standard) */}
              <label
                className={`flex items-start gap-2.5 p-3 rounded-lg border cursor-pointer transition ${
                  verdict === "PRINTABLE"
                    ? "border-emerald-500 bg-emerald-50/50 ring-1 ring-emerald-500"
                    : "border-slate-200 bg-white hover:border-slate-300"
                }`}
              >
                <input
                  type="radio"
                  name="verdict"
                  checked={verdict === "PRINTABLE"}
                  onChange={() => setVerdict("PRINTABLE")}
                  className="mt-0.5 accent-emerald-600"
                />
                <div className="text-xs">
                  <div className="font-semibold text-slate-900 flex items-center gap-1.5">
                    <CheckCircle2 className="h-3.5 w-3.5 text-emerald-600" />
                    Imprimable (Standard)
                  </div>
                  <p className="text-slate-500 mt-0.5">
                    Géométrie saine en PLA, envoi direct du devis au client.
                  </p>
                </div>
              </label>

              {/* Choix 2 : Imprimable avec améliorations */}
              <label
                className={`flex items-start gap-2.5 p-3 rounded-lg border cursor-pointer transition ${
                  verdict === "NEEDS_IMPROVEMENT"
                    ? "border-amber-500 bg-amber-50/50 ring-1 ring-amber-500"
                    : "border-slate-200 bg-white hover:border-slate-300"
                }`}
              >
                <input
                  type="radio"
                  name="verdict"
                  checked={verdict === "NEEDS_IMPROVEMENT"}
                  onChange={() => setVerdict("NEEDS_IMPROVEMENT")}
                  className="mt-0.5 accent-amber-600"
                />
                <div className="text-xs">
                  <div className="font-semibold text-slate-900 flex items-center gap-1.5">
                    <AlertTriangle className="h-3.5 w-3.5 text-amber-600" />
                    Imprimable avec améliorations
                  </div>
                  <p className="text-slate-500 mt-0.5">
                    Devis assorti de réserves ou proposition de RDV dessinateur.
                  </p>
                </div>
              </label>

              {/* Choix 3 : Non imprimable */}
              <label
                className={`flex items-start gap-2.5 p-3 rounded-lg border cursor-pointer transition ${
                  verdict === "NOT_PRINTABLE"
                    ? "border-rose-500 bg-rose-50/50 ring-1 ring-rose-500"
                    : "border-slate-200 bg-white hover:border-slate-300"
                }`}
              >
                <input
                  type="radio"
                  name="verdict"
                  checked={verdict === "NOT_PRINTABLE"}
                  onChange={() => setVerdict("NOT_PRINTABLE")}
                  className="mt-0.5 accent-rose-600"
                />
                <div className="text-xs">
                  <div className="font-semibold text-slate-900 flex items-center gap-1.5">
                    <XCircle className="h-3.5 w-3.5 text-rose-600" />
                    Non imprimable
                  </div>
                  <p className="text-slate-500 mt-0.5">
                    Pièce inadaptée à la FDM PLA, refus motivé + option RDV.
                  </p>
                </div>
              </label>
            </div>

            {/* Sous-options pour Imprimable avec améliorations */}
            {verdict === "NEEDS_IMPROVEMENT" && (
              <div className="space-y-3 rounded-lg bg-slate-50 p-3 border border-slate-200">
                <div>
                  <label className="block text-xs font-medium text-slate-700 mb-1">
                    Commentaire / pistes d&apos;amélioration *
                  </label>
                  <textarea
                    rows={3}
                    value={comment}
                    onChange={(e) => setComment(e.target.value)}
                    placeholder="Ex : Parois < 0.8 mm sur le clip supérieur, supports importants requis..."
                    className="w-full rounded-md border border-slate-300 bg-white px-2.5 py-2 text-xs focus:border-orange-500 focus:outline-none focus:ring-1 focus:ring-orange-500"
                  />
                </div>

                <div>
                  <span className="block text-xs font-medium text-slate-700 mb-1.5">
                    Suite proposée au client
                  </span>
                  <div className="grid grid-cols-2 gap-2">
                    <button
                      type="button"
                      onClick={() => setImprovementAction("QUOTE")}
                      className={`px-2.5 py-2 rounded-md text-xs font-medium border transition ${
                        improvementAction === "QUOTE"
                          ? "bg-slate-900 text-white border-slate-900"
                          : "bg-white text-slate-700 border-slate-300 hover:bg-slate-100"
                      }`}
                    >
                      Envoyer devis
                    </button>
                    <button
                      type="button"
                      onClick={() => setImprovementAction("DESIGNER_MEETING")}
                      className={`px-2.5 py-2 rounded-md text-xs font-medium border transition ${
                        improvementAction === "DESIGNER_MEETING"
                          ? "bg-slate-900 text-white border-slate-900"
                          : "bg-white text-slate-700 border-slate-300 hover:bg-slate-100"
                      }`}
                    >
                      Proposer RDV dessinateur
                    </button>
                  </div>
                </div>
              </div>
            )}

            {/* Synchronisation directe avec vos changements dans PrusaSlicer -> Prix TVAC */}
            {(verdict === "PRINTABLE" ||
              (verdict === "NEEDS_IMPROVEMENT" &&
                improvementAction === "QUOTE")) && (
              <div className="rounded-lg bg-slate-50 p-3.5 border border-slate-200 space-y-3">
                {syncedSlice ? (
                  <div className="rounded-md bg-emerald-50 border border-emerald-200 p-2.5 space-y-1.5 text-xs">
                    <div className="flex items-center justify-between">
                      <span className="font-semibold text-emerald-900 flex items-center gap-1.5">
                        <CheckCircle2 className="h-3.5 w-3.5 text-emerald-600" />
                        Synchronisé avec PrusaSlicer
                      </span>
                      <span className="text-[10px] font-mono text-emerald-800 bg-emerald-100 px-1.5 py-0.5 rounded">
                        PLA 1.24 g/cm³
                      </span>
                    </div>
                    <div className="font-mono text-[11px] text-emerald-950 space-y-0.5">
                      <div>
                        Poids extrait :{" "}
                        <strong>{syncedSlice.weightGrams} g</strong>
                        {syncedSlice.filamentMm
                          ? ` (${(syncedSlice.filamentMm / 1000).toFixed(2)} m)`
                          : ""}
                      </div>
                      <div className="text-emerald-800">
                        {[
                          syncedSlice.fillDensity
                            ? `Remplissage ${syncedSlice.fillDensity}`
                            : null,
                          syncedSlice.layerHeight
                            ? `Couche ${syncedSlice.layerHeight}mm`
                            : null,
                          syncedSlice.supports
                            ? "Supports : Oui"
                            : "Supports : Non",
                          syncedSlice.estimatedTime
                            ? `Durée ${syncedSlice.estimatedTime}`
                            : null,
                        ]
                          .filter(Boolean)
                          .join(" • ")}
                      </div>
                    </div>
                    <p className="text-[11px] text-emerald-700">
                      Si vous modifiez à nouveau vos réglages dans PrusaSlicer et
                      ré-exportez le G-code, ce calcul se mettra à jour tout seul.
                    </p>
                  </div>
                ) : (
                  <div className="rounded-md bg-orange-50/80 border border-orange-200 p-2.5 space-y-1.5 text-xs">
                    <div className="flex items-center justify-between">
                      <span className="font-semibold text-orange-950 flex items-center gap-1.5">
                        <Cpu className="h-3.5 w-3.5 text-orange-600" />
                        Synchronisation directe PrusaSlicer
                      </span>
                      <RefreshCw className="h-3 w-3 text-orange-500 animate-spin" />
                    </div>
                    <p className="text-[11px] text-orange-900 leading-relaxed">
                      Faites vos réglages dans <strong>PrusaSlicer au centre</strong>{" "}
                      (orientation, supports, remplissage…), cliquez sur{" "}
                      <strong>Trancher maintenant</strong> puis{" "}
                      <strong>Exporter le G-code</strong> (<kbd>Entrée</kbd>) : le
                      poids exact et le prix TVAC se rempliront automatiquement ici.
                    </p>
                  </div>
                )}

                <div>
                  <label className="block text-xs font-medium text-slate-700 mb-1">
                    Poids PLA en grammes *
                  </label>
                  <div className="relative">
                    <input
                      type="number"
                      step="0.01"
                      min="0.1"
                      value={weightInput}
                      onChange={(e) => setWeightInput(e.target.value)}
                      placeholder="Auto-rempli à l'export PrusaSlicer (ou saisie manuelle)"
                      className="w-full rounded-md border border-slate-300 bg-white px-3 py-2 pr-8 text-sm font-mono focus:border-orange-500 focus:outline-none focus:ring-1 focus:ring-orange-500"
                    />
                    <span className="absolute right-3 top-2 text-xs text-slate-400 font-mono">
                      g
                    </span>
                  </div>
                </div>

                <div className="rounded-md bg-white p-3 border border-slate-200 flex items-center justify-between">
                  <div>
                    <div className="text-xs font-medium text-slate-700 flex items-center gap-1">
                      <Euro className="h-3.5 w-3.5 text-orange-600" />
                      Prix TVAC calculé
                    </div>
                    <div className="text-[11px] text-slate-400 font-mono">
                      ({hasValidWeight ? parsedWeight : 0}g × 0,20 €) + 10,00 €
                    </div>
                  </div>
                  <div className="text-lg font-bold font-mono text-slate-900">
                    {computedPrice !== null
                      ? `${computedPrice.toFixed(2)} €`
                      : "—"}
                  </div>
                </div>

                {verdict === "PRINTABLE" && (
                  <div>
                    <label className="block text-xs font-medium text-slate-700 mb-1">
                      Note opérateur (optionnel)
                    </label>
                    <input
                      type="text"
                      value={comment}
                      onChange={(e) => setComment(e.target.value)}
                      placeholder="Auto-rempli avec vos paramètres PrusaSlicer"
                      className="w-full rounded-md border border-slate-300 bg-white px-2.5 py-1.5 text-xs focus:border-orange-500 focus:outline-none"
                    />
                  </div>
                )}
              </div>
            )}

            {/* Sous-options pour Non imprimable */}
            {verdict === "NOT_PRINTABLE" && (
              <div className="space-y-3 rounded-lg bg-rose-50/50 p-3 border border-rose-200">
                <div>
                  <label className="block text-xs font-medium text-slate-800 mb-1">
                    Raison du refus *
                  </label>
                  <textarea
                    rows={3}
                    value={comment}
                    onChange={(e) => setComment(e.target.value)}
                    placeholder="Ex : Modèle non-manifold (maillage ouvert) ou dimensions hors volume 250×210×220 mm..."
                    className="w-full rounded-md border border-slate-300 bg-white px-2.5 py-2 text-xs focus:border-rose-500 focus:outline-none focus:ring-1 focus:ring-rose-500"
                  />
                </div>

                <label className="flex items-center gap-2 text-xs text-slate-700 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={proposeMeetingOnReject}
                    onChange={(e) => setProposeMeetingOnReject(e.target.checked)}
                    className="rounded accent-orange-600"
                  />
                  Proposer un RDV avec un dessinateur 3D
                </label>
              </div>
            )}

            <button
              type="submit"
              disabled={submitting}
              className="w-full inline-flex items-center justify-center gap-2 rounded-lg bg-slate-900 hover:bg-slate-800 disabled:opacity-50 text-white text-xs font-semibold py-2.5 px-4 transition shadow-sm"
            >
              <Send className="h-4 w-4" />
              {submitting
                ? "Enregistrement…"
                : verdict === "NOT_PRINTABLE"
                ? "Confirmer le refus"
                : verdict === "NEEDS_IMPROVEMENT" &&
                  improvementAction === "DESIGNER_MEETING"
                ? "Proposer un RDV dessinateur"
                : "Envoyer le devis au client"}
            </button>
          </form>
        )}

        {/* ==================== STATUT : QUOTE_SENT ==================== */}
        {order.status === "QUOTE_SENT" && (
          <div className="space-y-4">
            <div className="rounded-lg bg-slate-50 border border-slate-200 p-3.5 space-y-2 text-xs">
              <div className="flex justify-between">
                <span className="text-slate-500">Diagnostic</span>
                <span className="font-medium text-slate-900">
                  {order.verdict ? VERDICT_LABELS[order.verdict] : "—"}
                </span>
              </div>
              <div className="flex justify-between">
                <span className="text-slate-500">Poids PLA</span>
                <span className="font-mono font-medium text-slate-900">
                  {order.weightGrams !== null ? `${order.weightGrams} g` : "—"}
                </span>
              </div>
              <div className="flex justify-between border-t border-slate-200 pt-2">
                <span className="font-medium text-slate-700">Devis TVAC</span>
                <span className="font-mono text-base font-bold text-slate-900">
                  {order.priceTvac !== null
                    ? `${order.priceTvac.toFixed(2)} €`
                    : "—"}
                </span>
              </div>
              {order.operatorComment && (
                <div className="pt-2 border-t border-slate-200 text-slate-600">
                  <span className="block font-medium text-slate-700 mb-0.5">
                    Commentaire :
                  </span>
                  {order.operatorComment}
                </div>
              )}
            </div>

            <div className="rounded-lg bg-sky-50 border border-sky-200 p-3 text-xs text-sky-900">
              En attente de l&apos;accord du client. En mode pilote local (sans
              Odoo), utilisez le bouton ci-dessous pour simuler l&apos;acceptation.
            </div>

            <button
              type="button"
              disabled={submitting}
              onClick={() => updateStatus("APPROVED")}
              className="w-full inline-flex items-center justify-center gap-2 rounded-lg bg-emerald-600 hover:bg-emerald-500 disabled:opacity-50 text-white text-xs font-semibold py-2.5 px-4 transition shadow-sm"
            >
              <UserCheck className="h-4 w-4" />
              Simuler validation client
            </button>

            <button
              type="button"
              disabled={submitting}
              onClick={() => updateStatus("TO_ANALYZE")}
              className="w-full inline-flex items-center justify-center gap-1.5 rounded-lg border border-slate-300 bg-white hover:bg-slate-50 text-slate-700 text-xs font-medium py-2 px-3 transition"
            >
              <RotateCcw className="h-3.5 w-3.5" />
              Modifier l&apos;analyse / le devis
            </button>
          </div>
        )}

        {/* ==================== STATUT : APPROVED ==================== */}
        {order.status === "APPROVED" && (
          <div className="space-y-4">
            <div className="rounded-lg bg-emerald-50/70 border border-emerald-200 p-3 text-xs space-y-1">
              <div className="font-semibold text-emerald-900">
                Devis validé ({order.priceTvac?.toFixed(2) ?? "—"} € TVAC •{" "}
                {order.weightGrams ?? "—"} g PLA)
              </div>
              {syncedSlice ? (
                <p className="text-emerald-800 font-mono text-[11px] break-all">
                  Prêt : {syncedSlice.filename} ({syncedSlice.weightGrams} g)
                </p>
              ) : (
                <p className="text-emerald-800">
                  Exportez le G-code depuis le{" "}
                  <strong>PrusaSlicer au centre</strong> puis lancez-le en un
                  clic sur la Prusa MK4.
                </p>
              )}
            </div>

            <button
              type="button"
              disabled={submitting}
              onClick={handleSendLatestExportedGcode}
              className="w-full inline-flex items-center justify-center gap-2 rounded-lg bg-orange-600 hover:bg-orange-500 disabled:opacity-50 text-white text-xs font-semibold py-2.5 px-3 transition shadow-sm"
            >
              <Zap className="h-4 w-4" />
              {submitting
                ? "Transfert vers PrusaLink…"
                : syncedSlice
                ? `Envoyer ${syncedSlice.filename} sur la Prusa`
                : "Lancer le dernier G-code exporté de PrusaSlicer"}
            </button>

            <div className="relative flex items-center py-1">
              <div className="flex-grow border-t border-slate-200" />
              <span className="shrink-0 px-2 text-[10px] uppercase tracking-wider text-slate-400">
                ou glisser un autre fichier
              </span>
              <div className="flex-grow border-t border-slate-200" />
            </div>

            {/* Drag & Drop G-Code */}
            <div
              onDragOver={(e) => {
                e.preventDefault();
                setIsDragging(true);
              }}
              onDragLeave={() => setIsDragging(false)}
              onDrop={(e) => {
                e.preventDefault();
                setIsDragging(false);
                handleSelectGcodeFile(e.dataTransfer.files?.[0]);
              }}
              onClick={() => fileInputRef.current?.click()}
              className={`border-2 border-dashed rounded-xl p-4 text-center cursor-pointer transition ${
                isDragging
                  ? "border-orange-500 bg-orange-50"
                  : gcodeFile
                  ? "border-emerald-400 bg-emerald-50/40"
                  : "border-slate-300 bg-slate-50 hover:border-orange-400 hover:bg-white"
              }`}
            >
              <input
                ref={fileInputRef}
                type="file"
                accept=".gcode,.bgcode"
                className="hidden"
                onChange={(e) => handleSelectGcodeFile(e.target.files?.[0])}
              />
              {gcodeFile ? (
                <div className="space-y-1">
                  <FileCode2 className="h-6 w-6 text-emerald-600 mx-auto" />
                  <p className="text-xs font-semibold text-slate-900 break-all">
                    {gcodeFile.name}
                  </p>
                  <p className="text-[11px] text-slate-500">
                    {(gcodeFile.size / 1024).toFixed(1)} Ko — Cliquez pour
                    remplacer
                  </p>
                </div>
              ) : (
                <div className="space-y-1">
                  <UploadCloud className="h-7 w-7 text-slate-400 mx-auto" />
                  <p className="text-xs font-medium text-slate-700">
                    Glissez-déposez un{" "}
                    <span className="font-mono">.gcode</span> /{" "}
                    <span className="font-mono">.bgcode</span>
                  </p>
                </div>
              )}
            </div>

            {gcodeFile && (
              <button
                type="button"
                disabled={submitting}
                onClick={handleUploadAndPrint}
                className="w-full inline-flex items-center justify-center gap-2 rounded-lg bg-slate-900 hover:bg-slate-800 disabled:opacity-50 text-white text-xs font-semibold py-2.5 px-4 transition shadow-sm"
              >
                <Printer className="h-4 w-4" />
                Envoyer ce fichier et lancer sur la Prusa
              </button>
            )}
          </div>
        )}

        {/* ==================== STATUT : PRINTING ==================== */}
        {order.status === "PRINTING" && (
          <div className="space-y-4">
            <div className="rounded-lg bg-orange-50 border border-orange-200 p-3.5 space-y-2 text-xs">
              <div className="font-semibold text-orange-900 flex items-center gap-2">
                <span className="h-2 w-2 rounded-full bg-orange-500 animate-pulse" />
                Impression PLA en cours sur la Prusa

function formatDuration(seconds: number | null): string {
  if (seconds === null || seconds < 0) return "—";
  if (seconds === 0) return "0s";
  const hrs = Math.floor(seconds / 3600);
  const mins = Math.floor((seconds % 3600) / 60);
  const secs = seconds % 60;
  if (hrs > 0) return `${hrs}h ${String(mins).padStart(2, "0")}m`;
  if (mins > 0) return `${mins}m ${String(secs).padStart(2, "0")}s`;
  return `${secs}s`;
}

import path from "node:path";

export interface ParsedGcodeMetadata {
  filename: string;
  fullPath: string;
  mtimeMs: number;
  weightGrams: number;
  filamentMm: number | null;
  filamentCm3: number | null;
  estimatedTime: string | null;
  fillDensity: string | null;
  layerHeight: string | null;
  supports: boolean;
}

const PLA_DENSITY = 1.24;

function matchValue(text: string, keyPattern: string): string | null {
  // Supporte à la fois le format ASCII G-code ("; key = value") et le format binaire .bgcode ("key=value")
  const regex = new RegExp(
    `(?:;\\s*)?${keyPattern}\\s*=\\s*([^\\r\\n\\x00]+)`,
    "i"
  );
  const m = regex.exec(text);
  return m ? m[1].trim() : null;
}

/**
 * Extrait le poids exact de PLA (en grammes), le temps estimé et les paramètres de tranchage
 * directement depuis un fichier `.gcode` (ASCII) ou `.bgcode` (binaire Prusa MK4) généré par PrusaSlicer.
 */
export async function parsePrusaGcodeMetadata(
  filePath: string
): Promise<ParsedGcodeMetadata | null> {
  const st = await fs.stat(filePath).catch(() => null);
  if (!st || !st.isFile() || st.size < 32) return null;

  const handle = await fs.open(filePath, "r");
  try {
    // Dans un .bgcode, les métadonnées sont dans les premiers Ko ;
    // dans un .gcode ASCII, elles sont dans les derniers Ko. On lit tête + queue.
    const headLen = Math.min(st.size, 65536);
    const headBuf = Buffer.alloc(headLen);
    await handle.read(headBuf, 0, headLen, 0);

    let combinedText = headBuf.toString("latin1");

    if (st.size > headLen) {
      const tailLen = Math.min(st.size - headLen, 65536);
      const tailBuf = Buffer.alloc(tailLen);
      await handle.read(tailBuf, 0, tailLen, st.size - tailLen);
      combinedText += "\n" + tailBuf.toString("latin1");
    }

    const totalGramsStr =
      matchValue(combinedText, "total filament used \\[g\\]") ??
      matchValue(combinedText, "filament used \\[g\\]");
    const mmStr = matchValue(combinedText, "filament used \\[mm\\]");
    const cm3Str = matchValue(combinedText, "filament used \\[cm3\\]");
    const estimatedTime = matchValue(
      combinedText,
      "estimated printing time \\(normal mode\\)"
    );
    const fillDensity = matchValue(combinedText, "fill_density");
    const layerHeight = matchValue(combinedText, "layer_height");
    const supportStr = matchValue(combinedText, "support_material");

    const filamentMm = mmStr ? parseFloat(mmStr) : null;
    const filamentCm3 = cm3Str ? parseFloat(cm3Str) : null;
    let weightGrams = totalGramsStr ? parseFloat(totalGramsStr) : NaN;

    if ((!Number.isFinite(weightGrams) || weightGrams <= 0) && filamentCm3 && filamentCm3 > 0) {
      weightGrams = filamentCm3 * PLA_DENSITY;
    } else if (
      (!Number.isFinite(weightGrams) || weightGrams <= 0) &&
      filamentMm &&
      filamentMm > 0
    ) {
      weightGrams = ((filamentMm * 2.40528) / 1000) * PLA_DENSITY;
    }

    if (!Number.isFinite(weightGrams) || weightGrams <= 0) {
      return null;
    }

    const supports =
      supportStr === "1" ||
      (typeof supportStr === "string" &&
        supportStr.toLowerCase() === "true");

    return {
      filename: path.basename(filePath),
      fullPath: filePath,
      mtimeMs: st.mtimeMs,
      weightGrams: Number(Math.max(0.1, weightGrams).toFixed(2)),
      filamentMm:
        filamentMm && Number.isFinite(filamentMm)
          ? Number(filamentMm.toFixed(1))
          : null,
      filamentCm3:
        filamentCm3 && Number.isFinite(filamentCm3)
          ? Number(filamentCm3.toFixed(2))
          : null,
      estimatedTime: estimatedTime || null,
      fillDensity: fillDensity || null,
      layerHeight: layerHeight || null,
      supports,
    };
  } finally {
    await handle.close();
  }
}

async function getPrusaSlicerLastOutputDir(): Promise<string | null> {

/**
 * Recherche le fichier `.gcode` ou `.bgcode` le plus récemment exporté par l'opérateur
 * depuis l'ouverture de la pièce dans PrusaSlicer. S'il a été enregistré dans un autre dossier,
 * il est automatiquement copié dans `./data/gcode/`.
 */
export async function findLatestExportedGcodeSince(
  sinceMs: number
): Promise<ParsedGcodeMetadata | null> {
  const gcodeDir = path.resolve(process.cwd(), "data", "gcode");
  await fs.mkdir(gcodeDir, { recursive: true });

  const dirsToCheck = new Set<string>([gcodeDir]);
  const lastDir = await getPrusaSlicerLastOutputDir();
  if (lastDir) {
    dirsToCheck.add(path.resolve(lastDir));
  }

  let newestCandidate: { fullPath: string; mtimeMs: number; inGcodeDir: boolean } | null =
    null;

  for (const dir of dirsToCheck) {
    const entries = await fs.readdir(dir).catch(() => []);
    for (const name of entries) {
      const ext = path.extname(name).toLowerCase();
      if (ext !== ".gcode" && ext !== ".bgcode") continue;
      const fullPath = path.join(dir, name);
      const st = await fs.stat(fullPath).catch(() => null);
      if (!st || !st.isFile()) continue;
      if (st.mtimeMs < sinceMs) continue;

      if (!newestCandidate || st.mtimeMs > newestCandidate.mtimeMs) {
        newestCandidate = {
          fullPath,
          mtimeMs: st.mtimeMs,
          inGcodeDir: dir === gcodeDir,
        };
      }
    }
  }

  if (!newestCandidate) return null;

  let targetPath = newestCandidate.fullPath;
  if (!newestCandidate.inGcodeDir) {
    const dest = path.join(gcodeDir, path.basename(newestCandidate.fullPath));
    await fs.copyFile(newestCandidate.fullPath, dest).catch(() => {});
    targetPath = dest;
  }

  const parsed = await parsePrusaGcodeMetadata(targetPath);
  if (!parsed) return null;
  // Conserve le timestamp exact d'export pour le suivi côté UI
  parsed.mtimeMs = newestCandidate.mtimeMs;
  return parsed;
}

function md5(value: string): string {
  return crypto.createHash("md5").update(value).digest("hex");
}

/**
 * Génère l'en-tête HTTP Digest Auth (RFC 2617 / RFC 2069) pour le firmware Buddy de la Prusa MK4
 * lorsque PrusaLink demande l'authentification Digest (`user: maker` + password PrusaLink).
 */
function buildDigestAuthHeader(params: {
  wwwAuthenticate: string;
  method: string;
  uriPath: string;
  username: string;
  password: string;
}): string | null {
  const { wwwAuthenticate, method, uriPath, username, password } = params;
  if (!/^Digest\s+/i.test(wwwAuthenticate)) return null;

  const extract = (key: string): string | undefined => {
    const quoted = new RegExp(`${key}="([^"]*)"`, "i").exec(wwwAuthenticate);
    if (quoted) return quoted[1];
    const unquoted = new RegExp(`${key}=([^,\\s]+)`, "i").exec(wwwAuthenticate);
    return unquoted?.[1];
  };

  const realm = extract("realm") ?? "Printer API";
  const nonce = extract("nonce");
  if (!nonce) return null;

  const qopRaw = extract("qop");
  const opaque = extract("opaque");

  const ha1 = md5(`${username}:${realm}:${password}`);
  const ha2 = md5(`${method.toUpperCase()}:${uriPath}`);

  const qopTokens = qopRaw
    ? qopRaw.split(",").map((s) => s.trim().toLowerCase())
    : [];
  const useAuthQop = qopTokens.includes("auth");

  let responseHash: string;
  const parts: string[] = [
    `username="${username}"`,
    `realm="${realm}"`,
    `nonce="${nonce}"`,
    `uri="${uriPath}"`,
  ];

  if (useAuthQop) {
    const nc = "00000001";
    const cnonce = crypto.randomBytes(8).toString("hex");
    responseHash = md5(`${ha1}:${nonce}:${nc}:${cnonce}:auth:${ha2}`);
    parts.push(`qop=auth`, `nc=${nc}`, `cnonce="${cnonce}"`);
  } else {
    responseHash = md5(`${ha1}:${nonce}:${ha2}`);
  }

  parts.push(`response="${responseHash}"`);
  if (opaque) {
    parts.push(`opaque="${opaque}"`);
  }

  return `Digest ${parts.join(", ")}`;
}

/**
 * Effectue une requête vers PrusaLink en envoyant `X-Api-Key`, et répond automatiquement
 * au challenge HTTP Digest (`maker` + mot de passe PrusaLink) si la Prusa MK4 renvoie 401.
 */
async function prusaFetch(
  host: string,
  uriPath: string,
  init: RequestInit & { timeoutMs?: number } = {}
): Promise<Response> {
  const apiKey = process.env.PRUSA_API_KEY?.trim() ?? "";
  const username = process.env.PRUSA_USER?.trim() || "maker";
  const timeoutMs = init.timeoutMs ?? 3500;
  const method = (init.method ?? "GET").toUpperCase();
  const url = `http://${host}${uriPath}`;

  const baseHeaders: Record<string, string> = {
    ...(init.headers as Record<string, string> | undefined),
    ...(apiKey ? { "X-Api-Key": apiKey } : {}),
  };

  const firstRes = await fetch(url, {
    ...init,
    method,
    headers: baseHeaders,
    cache: "no-store",
    signal: AbortSignal.timeout(timeoutMs),
  });

  if (firstRes.status !== 401 || !apiKey) {
    return firstRes;
  }

  const wwwAuth = firstRes.headers.get("www-authenticate") ?? "";
  const digestHeader = buildDigestAuthHeader({
    wwwAuthenticate: wwwAuth,
    method,
    uriPath,
    username,
    password: apiKey,
  });

  if (!digestHeader) {
    return firstRes;
  }

  const restHeaders: Record<string, string> = {
    ...(init.headers as Record<string, string> | undefined),
  };

  return fetch(url, {
    ...init,
    method,
    headers: {
      ...restHeaders,
      Authorization: digestHeader,
    },
    cache: "no-store",
    signal: AbortSignal.timeout(timeoutMs),
  });
}

function normalizePrusaState(rawState?: string): PrinterState {
  if (!rawState) return "OFFLINE";
  const upper = rawState.toUpperCase();
  if (upper === "PRINTING" || upper === "PAUSED" || upper === "BUSY") {
    return "PRINTING";
  }
  if (upper === "FINISHED") {
    return "FINISHED";
  }
  if (
    upper === "IDLE" ||
    upper === "READY" ||
    upper === "STOPPED" ||
    upper === "ATTENTION"
  ) {
    return "IDLE";
  }
  return "OFFLINE";
}

export async function fetchPrusaStatus(): Promise<PrinterStatusPayload> {
  const configuredHost =
    process.env.PRUSA_IP?.trim() || process.env.PRUSA_HOSTNAME?.trim() || "";

  if (isPrusaMockEnabled()) {
    const session = getMockSession();

          try {
            await updateOrderStatus(session.activeOrderId, "FINISHED");
          } catch {
            // Ignore
          }
        }
      } else {

          99,
          Math.max(1, Math.round((elapsedSec / session.durationSeconds) * 100))
        );

          1,
          Math.round(session.durationSeconds - elapsedSec)

          tempBed: 60.1,

          timeRemainingSeconds: remaining,
          activeFile: session.activeFile,

        activeFile: session.activeFile,

      state: "IDLE",
      rawState: "IDLE",

      targetNozzle: 0,
      tempBed: 23.2,

      progress: null,

      activeOrderId: null,

      timeRemainingSeconds: null,

        timeoutMs: 2500,

      const statusJob = statusData?.job ?? {};

      const rawState: string | undefined = printer.state ?? jobData?.state;
      const state = normalizePrusaState(rawState);

      const progressRaw = statusJob.progress ?? jobData?.progress ?? null;
      const progress =
        typeof progressRaw === "number" ? Math.round(progressRaw) : null;

      const timeRemainingRaw =
        statusJob.time_remaining ?? jobData?.time_remaining ?? null;
      const timeRemainingSeconds =
        typeof timeRemainingRaw === "number"
          ? Math.round(timeRemainingRaw)
          : null;

        mock: false,
        host,

        timeRemainingSeconds,
        activeFile,
      };

      lastErrorMsg = `Prusa MK4 (${host}) hors ligne ou sur un autre sous-réseau Wi-Fi`;
    }
  }

    state: "OFFLINE",
    mock: false,
    host: hosts[0],
    tempNozzle: null,
    targetNozzle: null,
    tempBed: null,
    targetBed: null,
    progress: null,
    timeRemainingSeconds: null,
    activeFile: null,
    message: lastErrorMsg,
  };

export async function uploadGcodeAndStartPrint(params: {
  orderId: string;
  originalFilename: string;
  fileBuffer: Buffer;
}) {
  const ext = path.extname(params.originalFilename).toLowerCase();
  if (ext !== ".gcode" && ext !== ".bgcode") {
    throw new Error("Seuls les fichiers .gcode et .bgcode sont acceptés.");
  }

  const base = path
    .basename(params.originalFilename, ext)
    .replace(/[^a-zA-Z0-9._-]/g, "_")
    .slice(0, 60);
  const suffix = `_${params.orderId.slice(-6)}`;
  const safeFilename = `${base || "print"}${
    base.endsWith(suffix) ? "" : suffix
  }${ext}`;

  const gcodeDir = path.join(process.cwd(), "data", "gcode");
  await fs.mkdir(gcodeDir, { recursive: true });
  const localPath = path.join(gcodeDir, safeFilename);
  await fs.writeFile(localPath, params.fileBuffer);

  if (isPrusaMockEnabled()) {
    const session = getMockSession();
    session.state = "PRINTING";
    session.activeFile = safeFilename;
    session.activeOrderId = params.orderId;
    session.startedAtMs = Date.now();

    const updatedOrder = await updateOrderStatus(params.orderId, "PRINTING", {
      gcodeFilename: safeFilename,
    });

    return {
      mock: true,
      filename: safeFilename,
      localPath: `./data/gcode/${safeFilename}`,
      order: updatedOrder,
    };
  }

  const hosts = getCandidateHosts();
  if (hosts.length === 0) {
    throw new Error("Variable PRUSA_IP manquante dans la configuration serveur.");
  }

  // Vérifie qu'une clé USB est bien présente dans la Prusa avant d'envoyer le fichier
  try {
    const storageRes = await prusaFetch(hosts[0], "/api/v1/storage", {
      headers: { Accept: "application/json" },
      timeoutMs: 3000,
    });
    if (storageRes.ok) {
      const storageData = await storageRes.json();
      const usb = (storageData?.storage_list ?? []).find(
        (s: { path?: string }) => s.path === "/usb/"
      );
      if (usb && usb.available === false) {
        throw new Error(
          "Aucune clé USB détectée dans la Prusa MK4. Insérez une clé USB (FAT32) à l'avant de l'imprimante puis réessayez."
        );
      }
      if (usb && usb.read_only === true) {
        throw new Error(
          "La clé USB de la Prusa est en lecture seule. Vérifiez la clé puis réessayez."
        );
      }
    }
  } catch (err) {
    if (err instanceof Error && err.message.includes("clé USB")) {
      throw err;
    }
    // Vérification non bloquante si l'endpoint storage ne répond pas
  }

  const uriPath = `/api/v1/files/usb/${encodeURIComponent(safeFilename)}`;
  let lastError: Error | null = null;

  for (const host of hosts) {
    try {
      const response = await prusaFetch(host, uriPath, {
        method: "PUT",
        headers: {
          "Content-Type": "application/octet-stream",
          "Content-Length": String(params.fileBuffer.byteLength),
          "Print-After-Upload": "?1",
          Overwrite: "?1",
        },
        body: new Uint8Array(params.fileBuffer),
        timeoutMs: 45000,
      });

      if (!response.ok) {
        const details = await response.text().catch(() => "");
        throw new Error(
          `Échec PrusaLink sur ${host} (HTTP ${response.status}) ${details}`.trim()
        );
      }

      const updatedOrder = await updateOrderStatus(params.orderId, "PRINTING", {
        gcodeFilename: safeFilename,
      });

      return {
        mock: false,
        host,
        filename: safeFilename,
        localPath: `./data/gcode/${safeFilename}`,
        order: updatedOrder,
      };
    } catch (err) {
      lastError =
        err instanceof Error
          ? err
          : new Error(`Impossible de joindre la Prusa sur ${host}`);
    }
  }

  throw (
    lastError ??
    new Error("Impossible d'envoyer le fichier G-code vers la Prusa MK4.")
  );
}

        timeout: 3000,
      }).catch(() => {});
    }
  })().finally(() => {
    runtime.startingPromise = null;
  });

  return runtime.startingPromise;
}

/**
 * - Ajuste la fenêtre principale de PrusaSlicer à 100% du cadre (0, 0, dispW, dispH).
 * - Contraint automatiquement toutes les boîtes de dialogue modales (ex: "Enregistrer le fichier G-code sous :")
 *   à rester entièrement visibles à l'intérieur du cadre (X >= 12, Y >= 12, largeur/hauteur <= cadre - 24),
 *   pour que les boutons "Annuler" / "Enregistrer" ne soient jamais coupés en haut ou à droite.
 */
export async function fitSlicerWindows(): Promise<void> {
  const env = { ...process.env, DISPLAY: DISPLAY_NUM };
  const { stdout: geomOut } = await execFileAsync(
    "xdotool",
    ["getdisplaygeometry"],
    { env, timeout: 1500 }
  );
  const [dispW, dispH] = geomOut.trim().split(/\s+/).map(Number);
  if (!dispW || !dispH) return;

  const { stdout: idsOut } = await execFileAsync(
    "xdotool",
    ["search", "--onlyvisible", "--class", "PrusaSlicer"],
    { env, timeout: 1500 }
  ).catch(() => ({ stdout: "" }));

  const ids = idsOut
    .trim()
    .split(/\s+/)
    .filter(Boolean);

  for (const wid of ids) {
    const { stdout: props } = await execFileAsync(
      "xprop",
      ["-id", wid, "WM_TRANSIENT_FOR", "_NET_WM_WINDOW_TYPE", "WM_NAME"],
      { env, timeout: 1500 }
    ).catch(() => ({ stdout: "" }));

    const isPopup =
      props.includes("_NET_WM_WINDOW_TYPE_MENU") ||
      props.includes("_NET_WM_WINDOW_TYPE_POPUP_MENU") ||
      props.includes("_NET_WM_WINDOW_TYPE_DROPDOWN_MENU") ||
      props.includes("_NET_WM_WINDOW_TYPE_COMBO") ||
      props.includes("_NET_WM_WINDOW_TYPE_TOOLTIP");

    if (isPopup) {
      continue;
    }

    const isDialog =
      (props.includes("WM_TRANSIENT_FOR(WINDOW)") &&
        !props.includes("WM_TRANSIENT_FOR:  not found")) ||
      props.includes("_NET_WM_WINDOW_TYPE_DIALOG");

    const { stdout: winGeom } = await execFileAsync(
      "xdotool",
      ["getwindowgeometry", "--shell", wid],
      { env, timeout: 1500 }
    ).catch(() => ({ stdout: "" }));

    const x = Number(/X=(-?\d+)/.exec(winGeom)?.[1] ?? 0);
    const y = Number(/Y=(-?\d+)/.exec(winGeom)?.[1] ?? 0);
    const w = Number(/WIDTH=(\d+)/.exec(winGeom)?.[1] ?? 0);
    const h = Number(/HEIGHT=(\d+)/.exec(winGeom)?.[1] ?? 0);

    if (w < 260 || h < 160) {
      continue;
    }

    if (isDialog) {
      const margin = 16;
      const maxW = Math.max(400, dispW - margin * 2);
      const maxH = Math.max(300, dispH - margin * 2);
      const targetW = Math.min(w, maxW);
      const targetH = Math.min(h, maxH);

      const needsResize = w > maxW || h > maxH;
      const outOfBounds =
        x < 4 ||
        y < 4 ||
        x + targetW > dispW - 4 ||
        y + targetH > dispH - 4;

      if (needsResize || outOfBounds) {
        if (needsResize) {
          await execFileAsync(
            "xdotool",
            ["windowsize", wid, String(targetW), String(targetH)],
            { env, timeout: 1500 }
          ).catch(() => {});
        }

        // Relit la taille réelle après application des contraintes GTK éventuelles
        const { stdout: afterGeom } = await execFileAsync(
          "xdotool",
          ["getwindowgeometry", "--shell", wid],
          { env, timeout: 1500 }
        ).catch(() => ({ stdout: winGeom }));

        const realW = Number(/WIDTH=(\d+)/.exec(afterGeom)?.[1] ?? targetW);
        const realH = Number(/HEIGHT=(\d+)/.exec(afterGeom)?.[1] ?? targetH);

        // Garantit toujours Y >= 0 (et X >= 0) pour que la barre supérieure GTK ("Enregistrer" / "Annuler") reste 100% visible
        const newX = Math.max(0, Math.floor((dispW - realW) / 2));
        const newY = Math.max(0, Math.floor((dispH - realH) / 2));

        await execFileAsync(
          "xdotool",
          ["windowmove", wid, String(newX), String(newY)],
          { env, timeout: 1500 }
        ).catch(() => {});
      }
      continue;
    }

    // Fenêtre principale PrusaSlicer -> plein cadre
    if (x !== 0 || y !== 0 || Math.abs(w - dispW) > 4 || Math.abs(h - dispH) > 4) {
      await execFileAsync(
        "xdotool",
        [
          "windowmove",
          wid,
          "0",
          "0",
          "windowsize",
          wid,
          String(dispW),
          String(dispH),
        ],
        { env, timeout: 1500 }
      ).catch(() => {});
    }
  }
}

async function killRunningPrusaSlicerOnDisplay(): Promise<void> {
  const runtime = getRuntime();
  if (runtime.slicerProc && !runtime.slicerProc.killed) {

    runtime.slicerProc = null;
  }

  await execFileAsync("flatpak", ["kill", "com.prusa3d.PrusaSlicer"], {

  }).catch(() => {});

  await new Promise((r) => setTimeout(r, 250));
}

export async function loadOrderInEmbeddedPrusaSlicer(params: {
  orderId?: string;
  stlAbsolutePath: string;
  stlFilename: string;
  forceReload?: boolean;
}): Promise<{ reloaded: boolean }> {
  await ensureDisplay();

  const runtime = getRuntime();

  const gcodeDir = path.resolve(process.cwd(), "data", "gcode");
  await fs.mkdir(gcodeDir, { recursive: true });

  const cleanName = params.stlFilename.replace(/[^a-zA-Z0-9._-]/g, "_");
  const workStlPath = path.join(gcodeDir, cleanName);
  await fs.copyFile(params.stlAbsolutePath, workStlPath);

  const env = { ...process.env, DISPLAY: DISPLAY_NUM };
  const { stdout: existingWin } = await execFileAsync(
    "xdotool",
    ["search", "--onlyvisible", "--class", "PrusaSlicer"],
    { env, timeout: 1500 }
  ).catch(() => ({ stdout: "" }));

  const isWindowAlive = Boolean(existingWin.trim());

  if (
    !params.forceReload &&
    isWindowAlive &&
    runtime.currentStlPath === params.stlAbsolutePath
  ) {
    if (params.orderId && runtime.activeOrderId !== params.orderId) {
      runtime.activeOrderId = params.orderId;
      runtime.sessionOpenedAtMs = Date.now();
    }
    await fitSlicerWindows().catch(() => {});
    return { reloaded: false };
  }

  await killRunningPrusaSlicerOnDisplay();

  const workspaceDir = process.cwd();
  const cleanEnv: NodeJS.ProcessEnv = {
    ...process.env,
    DISPLAY: DISPLAY_NUM,
    GDK_BACKEND: "x11",
    WAYLAND_DISPLAY: "",
  };
  delete cleanEnv.GIO_EXTRA_MODULES;
  delete cleanEnv.GTK_PATH;
  delete cleanEnv.GTK_IM_MODULE_FILE;

  const proc = spawn(
    "flatpak",
    [
      "run",
      "--nosocket=wayland",
      "--socket=x11",
      `--env=DISPLAY=${DISPLAY_NUM}`,
      "--env=GDK_BACKEND=x11",
      "--unset-env=WAYLAND_DISPLAY",
      "--unset-env=GIO_EXTRA_MODULES",
      "--unset-env=GTK_PATH",
      "--unset-env=GTK_IM_MODULE_FILE",
      `--filesystem=${workspaceDir}`,
      "--filesystem=/tmp",
      "com.prusa3d.PrusaSlicer",
      workStlPath,
    ],
    {
      env: cleanEnv,
      stdio: "ignore",
      detached: true,
    }
  );
  proc.unref();
  runtime.slicerProc = proc;
  runtime.currentStlPath = params.stlAbsolutePath;
  runtime.activeOrderId = params.orderId ?? runtime.activeOrderId;
  runtime.sessionOpenedAtMs = Date.now();

  for (const delay of [1500, 3000, 4800]) {
    setTimeout(() => {
      fitSlicerWindows().catch(() => {});
    }, delay);
  }

  return { reloaded: true };
}

/**
 * Détecte automatiquement l'exécutable réel de PrusaSlicer sur le PC Linux.
 */
export async function detectPrusaSlicer(): Promise<SlicerExecutable | null> {
  const customBin = process.env.PRUSA_SLICER_BIN?.trim();
  if (customBin) {
    try {
      await fs.access(customBin);
      return {
        kind: "native",
        command: customBin,
        baseArgs: [],
        label: path.basename(customBin),
      };
    } catch {
      // Continue vers l'auto-détection
    }
  }

  for (const bin of ["prusa-slicer", "prusaslicer"]) {
    if (await commandExists(bin)) {
      return {
        kind: "native",
        command: bin,
        baseArgs: [],
        label: bin,
      };
    }
  }

  if (await hasFlatpakPrusaSlicer()) {
    const workspaceDir = process.cwd();
    return {
      kind: "flatpak",
      command: "flatpak",
      baseArgs: [
        "run",
        "--unset-env=GIO_EXTRA_MODULES",
        "--unset-env=GTK_PATH",
        "--unset-env=GTK_IM_MODULE_FILE",
        `--filesystem=${workspaceDir}`,
        "--filesystem=/tmp",
        "com.prusa3d.PrusaSlicer",
      ],
      label: "PrusaSlicer (Flatpak officiel)",
    };
  }

  return null;
}

/**
 * Exécute le vrai moteur de tranchage PrusaSlicer en ligne de commande (`--export-gcode`)
 * exclusivement en PLA (1.24 g/cm³) pour calculer le poids exact en grammes et la durée estimée.
 */
export async function analyzeStlWithPrusaSlicer(params: {
  stlAbsolutePath: string;
  infillPercent?: number;
  supports?: boolean;
}): Promise<SlicerAnalysisResult> {
  const slicer = await detectPrusaSlicer();
  if (!slicer) {
    throw new Error(
      "PrusaSlicer n'est pas détecté sur ce PC. Impossible de lancer l'analyse automatique."
    );
  }

  const infill = Math.min(
    100,
    Math.max(5, Math.round(params.infillPercent ?? 15))
  );
  const supports = Boolean(params.supports);

  const tmpGcode = path.join(
    os.tmpdir(),
    `prusapilot_slice_${Date.now()}_${Math.random().toString(36).slice(2, 8)}.gcode`
  );

  try {
    let dimensionsMm: { x: number; y: number; z: number } | null = null;
    try {
      const { stdout: infoOut } = await execFileAsync(
        slicer.command,
        [...slicer.baseArgs, "--info", params.stlAbsolutePath],
        { timeout: 15000 }
      );
      const sx = /size_x\s*=\s*([\d.]+)/.exec(infoOut);
      const sy = /size_y\s*=\s*([\d.]+)/.exec(infoOut);
      const sz = /size_z\s*=\s*([\d.]+)/.exec(infoOut);
      if (sx && sy && sz) {
        dimensionsMm = {
          x: Number(parseFloat(sx[1]).toFixed(1)),
          y: Number(parseFloat(sy[1]).toFixed(1)),
          z: Number(parseFloat(sz[1]).toFixed(1)),
        };
      }
    } catch {
      // Non bloquant
    }

    const sliceArgs = [
      ...slicer.baseArgs,
      "--export-gcode",
      "--center",
      "125,105",
      "--nozzle-diameter",
      "0.4",
      "--layer-height",
      "0.2",
      "--first-layer-height",
      "0.2",
      "--perimeters",
      "2",
      "--top-solid-layers",
      "4",
      "--bottom-solid-layers",
      "4",
      "--fill-density",
      `${infill}%`,
      "--filament-diameter",
      "1.75",
      "--filament-density",
      String(PLA_DENSITY),
      ...(supports ? ["--support-material"] : []),
      "--output",
      tmpGcode,
      params.stlAbsolutePath,
    ];

    await execFileAsync(slicer.command, sliceArgs, {
      timeout: 60000,
      maxBuffer: 10 * 1024 * 1024,
    });

    const handle = await fs.open(tmpGcode, "r");
    try {
      const stat = await handle.stat();
      const tailSize = Math.min(stat.size, 65536);
      const buffer = Buffer.alloc(tailSize);
      await handle.read(buffer, 0, tailSize, Math.max(0, stat.size - tailSize));
      const tailText = buffer.toString("utf-8");

      const gramsMatch =
        /;\s*total filament used \[g\]\s*=\s*([\d.]+)/i.exec(tailText) ??
        /;\s*filament used \[g\]\s*=\s*([\d.]+)/i.exec(tailText);

      const mmMatch = /;\s*filament used \[mm\]\s*=\s*([\d.]+)/i.exec(tailText);
      const cm3Match = /;\s*filament used \[cm3\]\s*=\s*([\d.]+)/i.exec(
        tailText
      );
      const timeMatch =
        /;\s*estimated printing time \(normal mode\)\s*=\s*([^\r\n]+)/i.exec(
          tailText
        );

      const filamentMm = mmMatch
        ? Number(parseFloat(mmMatch[1]).toFixed(1))
        : null;
      const filamentCm3 = cm3Match
        ? Number(parseFloat(cm3Match[1]).toFixed(2))
        : null;

      let weightGrams: number | null = gramsMatch
        ? parseFloat(gramsMatch[1])
        : null;

      if ((!weightGrams || weightGrams <= 0) && filamentCm3 && filamentCm3 > 0) {
        weightGrams = filamentCm3 * PLA_DENSITY;
      } else if (
        (!weightGrams || weightGrams <= 0) &&
        filamentMm &&
        filamentMm > 0
      ) {
        weightGrams = ((filamentMm * 2.40528) / 1000) * PLA_DENSITY;
      }

      if (!weightGrams || weightGrams <= 0) {
        throw new Error(
          "PrusaSlicer a terminé le tranchage mais le poids de filament PLA n'a pas pu être extrait."
        );
      }

      return {
        weightGrams: Number(Math.max(0.1, weightGrams).toFixed(2)),
        filamentMm,
        filamentCm3,
        estimatedTime: timeMatch ? timeMatch[1].trim() : null,
        dimensionsMm,
        material: "PLA",
        infillPercent: infill,
        supports,
        slicerLabel: slicer.label,
      };
    } finally {
      await handle.close();
    }
  } finally {
    await fs.unlink(tmpGcode).catch(() => {});
  }
}

    needsDesignerMeeting: record.needsDesignerMeeting,

/**
 * Enregistre la décision technique de l'opérateur suite à l'inspection du STL.
 * Calcule automatiquement le prix TVAC côté serveur pour garantir la cohérence.
 */
export async function updateOrderDecision(
  id: string,
  input: UpdateOrderDecisionInput
): Promise<PrintOrder> {
  let nextStatus: OrderStatus;
  let weightGrams: number | null = null;
  let priceTvac: number | null = null;
  let needsDesignerMeeting = Boolean(input.needsDesignerMeeting);
  const comment = input.operatorComment?.trim() || null;

  if (input.verdict === "PRINTABLE") {
    if (typeof input.weightGrams !== "number" || input.weightGrams <= 0) {
      throw new Error("Le poids en grammes doit être supérieur à 0.");
    }
    weightGrams = Number(input.weightGrams.toFixed(2));
    priceTvac = calculatePriceTvac(weightGrams);
    needsDesignerMeeting = false;
    nextStatus = "QUOTE_SENT";
  } else if (input.verdict === "NEEDS_IMPROVEMENT") {
    if (input.improvementAction === "DESIGNER_MEETING") {
      needsDesignerMeeting = true;
      nextStatus = "REJECTED";
    } else {
      if (typeof input.weightGrams !== "number" || input.weightGrams <= 0) {
        throw new Error("Le poids en grammes est requis pour envoyer un devis.");
      }
      weightGrams = Number(input.weightGrams.toFixed(2));
      priceTvac = calculatePriceTvac(weightGrams);
      needsDesignerMeeting = false;
      nextStatus = "QUOTE_SENT";
    }
  } else {
    // NOT_PRINTABLE
    nextStatus = "REJECTED";
  }

  const record = await prisma.printRequest.update({
    where: { id },
    data: {
      verdict: input.verdict,
      weightGrams,
      priceTvac,
      operatorComment: comment,
      needsDesignerMeeting,
      status: nextStatus,
    },
  });

  return toPrintOrder(record);
}

 */
export async function updateOrderStatus(
  id: string,
  status: OrderStatus,
  options?: {
    gcodeFilename?: string;
    invoiced?: boolean;
  }
): Promise<PrintOrder> {
  const record = await prisma.printRequest.update({

    data: {

        ? { gcodeFilename: options.gcodeFilename }
        : {}),

        ? {
            invoiced: options.invoiced,
            invoicedAt: options.invoiced ? new Date() : null,

        : {}),

  });

const wss = new WebSocketServer({
  noServer: true,
  handleProtocols: (protocols) =>
    protocols.has("binary") ? "binary" : false,

const server = http.createServer(async (req, res) => {
  const urlPath = (req.url || "/").split("?")[0];

  if (urlPath === "/slicer-embed") {
    res.writeHead(200, {
      "Content-Type": "text/html; charset=utf-8",
      "Cache-Control": "no-store",
    });
    res.end(EMBED_HTML);
    return;
  }

  if (urlPath.startsWith("/slicer-novnc/")) {
    const relPath = urlPath.slice("/slicer-novnc/".length);
    const absPath = path.resolve(novncRoot, relPath);
    if (!absPath.startsWith(novncRoot + path.sep)) {
      res.writeHead(403);
      res.end("Forbidden");
      return;
    }
    try {
      const content = await fs.readFile(absPath);
      const ext = path.extname(absPath).toLowerCase();
      res.writeHead(200, {
        "Content-Type": MIME_TYPES[ext] || "application/octet-stream",
        "Cache-Control": "public, max-age=3600",
      });
      res.end(content);
      return;
    } catch {
      res.writeHead(404);
      res.end("Not found");
      return;
    }
  }

  handle(req, res);
});

server.on("upgrade", (req, socket, head) => {
  const urlPath = (req.url || "").split("?")[0];

  if (urlPath === "/slicer-ws") {
    wss.handleUpgrade(req, socket, head, (ws) => {
      const vncSocket = net.createConnection({
        host: "127.0.0.1",
        port: VNC_PORT,
      });

      vncSocket.on("data", (chunk) => {
        if (ws.readyState === ws.OPEN) {
          ws.send(chunk, { binary: true });
        }
      });

      ws.on("message", (data) => {
        if (!vncSocket.destroyed) {
          if (Buffer.isBuffer(data)) {
            vncSocket.write(data);
          } else if (data instanceof ArrayBuffer) {
            vncSocket.write(Buffer.from(data));
          } else if (Array.isArray(data)) {
            vncSocket.write(Buffer.concat(data));
          }
        }
      });

      const cleanup = () => {
        if (!vncSocket.destroyed) vncSocket.destroy();
        if (
          ws.readyState === ws.OPEN ||
          ws.readyState === ws.CONNECTING
        ) {
          ws.close();
        }
      };

      vncSocket.on("close", cleanup);
      vncSocket.on("error", cleanup);
      ws.on("close", cleanup);
      ws.on("error", cleanup);
    });
    return;
  }

  upgradeHandler(req, socket, head);
});

server.listen(port, () => {
  console.log(`> Prusa Pilot prêt sur http://${hostname}:${port}`);
});

# Prusa Pilot 3D — Console Opérateur Locale

Application web locale (Next.js App Router, TypeScript, Tailwind CSS, Prisma + SQLite, Three.js) pour piloter un service d'impression 3D sur une imprimante Prusa connectée en Wi-Fi via **PrusaLink**.

## Architecture évolutive (Prêt pour Odoo Online)

Toute la logique métier et d'accès aux demandes est isolée dans `src/services/orders.ts` :
- `getOrders`
- `getOrderById`
- `createOrder`
- `updateOrderDecision`
- `updateOrderStatus`

L'interface et les routes API consomment exclusivement ces fonctions et le DTO `PrintOrder` (`src/types/orders.ts`), ce qui permettra de remplacer ultérieurement SQLite par des appels JSON-RPC/XML-RPC vers **Odoo Online** sans modifier les composants UI.

## Stockage local

- Base de données SQLite : `prisma/dev.db`
- Modèles 3D clients (`.stl`) : `./data/stl/`
- Fichiers tranchés PrusaSlicer (`.gcode`, `.bgcode`) : `./data/gcode/`

## Configuration (`.env`)

Copiez `.env.example` vers `.env` et ajustez selon votre environnement :

```bash
cp .env.example .env
```

- `PRUSA_MOCK="true"` : simule localement les réponses de l'imprimante Prusa (températures, progression, lancement G-code) hors du réseau du laboratoire.
- `PRUSA_MOCK="false"` : active la communication Wi-Fi réelle avec PrusaLink (`PRUSA_IP` + header `X-Api-Key: PRUSA_API_KEY`).

## Commandes utiles

```bash
# Installer les dépendances
npm install

# Appliquer les migrations SQLite
npx prisma migrate dev --name init

# Générer les fichiers STL de test et les 5 demandes fictives
npm run db:seed

# Lancer le serveur de développement local
npm run dev
```

Ouvrez ensuite `http://localhost:3000`.
